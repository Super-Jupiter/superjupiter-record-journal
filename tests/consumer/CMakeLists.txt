cmake_minimum_required(VERSION 3.16)
project(SuperJupiterRecordJournalConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_RECORD_JOURNAL_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_RECORD_JOURNAL_SOURCE_DIR}" record-journal)
  if(TARGET superjupiter-record-journal-tests OR TARGET superjupiter-record-journal-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterRecordJournal 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-record-journal-consumer main.cpp)
target_link_libraries(superjupiter-record-journal-consumer PRIVATE SuperJupiter::RecordJournal)
